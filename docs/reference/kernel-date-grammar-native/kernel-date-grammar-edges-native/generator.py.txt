"""Independent native evidence for existing rejection assumptions and grammar edges."""
from pathlib import Path
import re,json,subprocess,hashlib
root=Path(__file__).resolve().parents[1];dest=root/'target/kernel-date-grammar-edges-native';dest.mkdir(exist_ok=False)
original=json.loads((root/'target/kernel-dates-expanded-native-0.1.25/capture.json').read_text())
oracle=root/'target/frame-kernel-errors-native-0.1.9-extended/kernel-pool-oracle.exe'
assert hashlib.sha256(oracle.read_bytes()).hexdigest()==original['oracle_sha256']
test=(root.parent/'STKCraft-time/tests/kernel_dates.rs').read_text()
body=test.split('fn malformed_literals_are_rejected_not_approximated()',1)[1].split('] {',1)[0]
literals=re.findall(r'"(@[^"\n]*|2000-JAN-01)"',body)
literals += ['@2000-JAN-1.','@2000-JAN-1.5','@2000-JAN-1E0','@2000-00-1','@2000-24-1',
 '@2000-99-1','@2000-JAN-99','@2000-JAN--1','@-1-JAN-1','@0000-JAN-1',
 '@2000-FE-1','@2000-J-1','@2000-JANUAR-1','@2000-FEBRUAR-1','@2000-SEPTEMBE-1',
 '@2000-NOVEMBE-1','@2000-JANUARYX-1','@2000-JANUARY-1',
 '@JD2451545.123456789','@2451545.123456789JD','@jd2451545.123456789',
 '@JED2451545','@MJD51544.5','@2451545.0','@JD-1.25','@JD0.0',
 '@2000-001T12:00:00.123456','@2000-366T23:59:59.999999',
 '@2001-366T00:00:00','@2000-000T00:00:00','@2000-999T00:00:00',
 '@2000-060-12:00:00','@2000-060/12:00:00',
 '@2000/JAN/01-12:00:00','@2000-JAN/01/12:00:00',
 '@2000-JAN-01-12:00:60.5','@2000-JAN-01-100:200:300.123',
 '@2000-JAN-01-12:00:.','@2000-JAN-01-12:00:.5','@2000-JAN-01-12:00:5.',
 '@2000-JAN-01-12:00:+5','@2000-JAN-01-12:00:-5']
literals=list(dict.fromkeys(literals));commands=[]
for index,literal in enumerate(literals):
 path=dest/f'date-{index:03}.tf';path.write_bytes(('KPL/FK\n\\begindata\nEPOCH='+literal+'\n\\begintext\n').encode('ascii'))
 commands.extend(['C','L|'+str(path),'D|EPOCH'])
data=('\n'.join(commands)+'\n').encode('ascii');native=subprocess.run([str(oracle)],input=data,capture_output=True)
assert native.returncode==0 and native.stderr.startswith(b'CRT_FMA:1\r\n')
for name,raw in [('stdin',data),('stdout',native.stdout),('stderr',native.stderr)]: (dest/('native-'+name+'.bin')).write_bytes(raw)
replies=native.stdout.decode('ascii').splitlines();assert len(replies)==len(commands)
records=[dict(literal=literal,load=replies[3*i+1],value=replies[3*i+2]) for i,literal in enumerate(literals)]
(dest/'dates.json').write_text(json.dumps(records,indent=2)+'\n',encoding='ascii')
print(json.dumps(dict(cases=len(records),native_accepted=sum(r['load']=='OK' for r in records))))
